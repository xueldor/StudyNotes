import android.os.RemoteCallbackList;
import android.view.IHsaeKeyEventListener;

private RemoteCallbackList<IHsaeKeyEventListener> mHsaeKeyEventListeners = new RemoteCallbackList<IHsaeKeyEventListener>();
private WindowManagerPolicy.HsaeKeyEventListener mHsaeKeyEventListenerInternal;

+    @Override
+    public void registerHsaeKeyEventListener(IHsaeKeyEventListener listener) {
+        Log.d(TAG, "register hsaeKeyEvent listener from " + Binder.getCallingPid());
+        mHsaeKeyEventListeners.register(listener);
+        if (mHsaeKeyEventListenerInternal == null) {
+            mHsaeKeyEventListenerInternal = new WindowManagerPolicy.HsaeKeyEventListener() {
+                @Override
+                public void onHsaeKeyEvent(int keyCode, boolean isDown) {
+                    if (null == mHsaeKeyEventListeners) {
+                        return;
+                    }
+                    int i = mHsaeKeyEventListeners.beginBroadcast();
+                    while (i > 0) {
+                        i--;
+                        IHsaeKeyEventListener listener = mHsaeKeyEventListeners
+                                .getBroadcastItem(i);
+                        try {
+                            listener.onKeyEventChanged(keyCode, isDown);
+                        } catch (RemoteException e) {
+                            // Nothing to do in here, RemoteCallbackListener will
+                            // clean it up.
+                        }
+                    }
+                    mHsaeKeyEventListeners.finishBroadcast();
+
+                }
+            };
+            mPolicy.registerHsaeKeyEventListener(mHsaeKeyEventListenerInternal);
+        }
+    }
